# Task: Live Preview JavaScript

## Description
Create JavaScript that loads the custom CSS on page load and enables live preview when changing colors in settings. Uses OWL component or simple DOM manipulation to inject/update CSS variables.

## Acceptance Criteria
- [ ] CSS loaded from controller on page load
- [ ] Color changes preview immediately (no page reload)
- [ ] No flash of unstyled content (FOUC)
- [ ] Works for all users (not just admin)
- [ ] Performance: <100ms for color updates
- [ ] No console errors

## Technical Details

### File: `static/src/js/color_customizer.js`
```javascript
/** @odoo-module **/

import { registry } from "@web/core/registry";
import { useService } from "@web/core/utils/hooks";
import { Component, onWillStart, onMounted } from "@odoo/owl";

const COLOR_CSS_ID = 'color-customizer-theme';
const CSS_ENDPOINT = '/color_customizer/theme.css';

/**
 * Load custom color CSS on application start
 */
async function loadColorTheme() {
    // Check if already loaded
    if (document.getElementById(COLOR_CSS_ID)) {
        return;
    }

    try {
        const response = await fetch(CSS_ENDPOINT);
        if (response.ok) {
            const css = await response.text();
            const style = document.createElement('style');
            style.id = COLOR_CSS_ID;
            style.textContent = css;
            document.head.appendChild(style);
        }
    } catch (error) {
        console.warn('Color customizer: Could not load theme CSS', error);
    }
}

/**
 * Update CSS variables in real-time
 * @param {string} color - Hex color value
 */
function updateLivePreview(color) {
    if (!color || !color.match(/^#[0-9A-Fa-f]{6}$/)) {
        return;
    }

    const root = document.documentElement;
    root.style.setProperty('--custom-primary', color);
    root.style.setProperty('--custom-primary-hover', darkenColor(color, 0.1));
    root.style.setProperty('--custom-primary-active', darkenColor(color, 0.2));
    root.style.setProperty('--custom-primary-light', lightenColor(color, 0.85));
    root.style.setProperty('--custom-primary-text', getContrastColor(color));
}

/**
 * Color manipulation utilities
 */
function hexToRgb(hex) {
    const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    return result ? {
        r: parseInt(result[1], 16),
        g: parseInt(result[2], 16),
        b: parseInt(result[3], 16)
    } : null;
}

function rgbToHex(r, g, b) {
    return '#' + [r, g, b].map(x => {
        const hex = Math.max(0, Math.min(255, Math.round(x))).toString(16);
        return hex.length === 1 ? '0' + hex : hex;
    }).join('');
}

function darkenColor(hex, amount) {
    const rgb = hexToRgb(hex);
    if (!rgb) return hex;
    const factor = 1 - amount;
    return rgbToHex(rgb.r * factor, rgb.g * factor, rgb.b * factor);
}

function lightenColor(hex, amount) {
    const rgb = hexToRgb(hex);
    if (!rgb) return hex;
    return rgbToHex(
        rgb.r + (255 - rgb.r) * amount,
        rgb.g + (255 - rgb.g) * amount,
        rgb.b + (255 - rgb.b) * amount
    );
}

function getContrastColor(hex) {
    const rgb = hexToRgb(hex);
    if (!rgb) return '#ffffff';
    const luminance = (0.299 * rgb.r + 0.587 * rgb.g + 0.114 * rgb.b) / 255;
    return luminance < 0.5 ? '#ffffff' : '#000000';
}

/**
 * Refresh theme CSS from server
 */
async function refreshThemeCSS() {
    const existingStyle = document.getElementById(COLOR_CSS_ID);
    if (existingStyle) {
        existingStyle.remove();
    }
    // Clear inline styles
    document.documentElement.style.removeProperty('--custom-primary');
    document.documentElement.style.removeProperty('--custom-primary-hover');
    document.documentElement.style.removeProperty('--custom-primary-active');
    document.documentElement.style.removeProperty('--custom-primary-light');
    document.documentElement.style.removeProperty('--custom-primary-text');

    await loadColorTheme();
}

// Load theme CSS on application start
registry.category("services").add("colorCustomizer", {
    start() {
        loadColorTheme();
        return {
            updateLivePreview,
            refreshThemeCSS,
        };
    },
});

// Also expose globally for settings page color picker
window.ColorCustomizer = {
    updateLivePreview,
    refreshThemeCSS,
};

// Listen for color input changes on settings page
document.addEventListener('input', (event) => {
    if (event.target.name === 'primary_color' && event.target.type === 'color') {
        updateLivePreview(event.target.value);
    }
});
```

### Integration Points
1. **Service Registry**: Registers as Odoo service for access from other components
2. **Global Object**: `window.ColorCustomizer` for settings page integration
3. **Event Listener**: Catches color input changes for live preview
